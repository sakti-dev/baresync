# React

This page shows the canonical React integration pattern used in the inventory example app. It uses React Query (`@tanstack/react-query`) for data fetching and Tauri events for cache invalidation.

## Setup

### 1. Install dependencies

```bash
bun add @tanstack/react-query baresync @tauri-apps/api drizzle-orm
```

### 2. Create the database

```tsx
import { invoke } from "@tauri-apps/api/core";
import { createTauriDrizzleDatabase } from "baresync/db";
import { items, locations } from "@your-app/sync-contract/local-synced-schema";
import { syncOutbox, syncCursors } from "@your-app/sync-contract/local-schema";

export const TABLE = { items, locations, syncOutbox, syncCursors };

export const db = createTauriDrizzleDatabase({ schema: TABLE, invoke });
```

### 3. Create the provider and hooks

This is the core integration. It does three things:

- Creates a `SyncClient` and provides it via React context
- Starts background polling on mount
- Listens for Tauri events and invalidates the React Query cache

```tsx
import { type QueryKey, useQuery, useQueryClient } from "@tanstack/react-query";
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";
import { createSyncClient, type SyncClient } from "baresync/tauri";
import {
  createContext,
  type ReactNode,
  useContext,
  useEffect,
  useRef,
  useState,
} from "react";

const SyncClientContext = createContext<SyncClient | null>(null);

export function SyncClientProvider({ children }: { children: ReactNode }) {
  const queryClient = useQueryClient();
  const [client] = useState(() =>
    createSyncClient({
      apiUrl: "http://127.0.0.1:18181",
      encoding: "json",
      scopeId: "your-scope-id",
      invoke,
    })
  );
  const initialized = useRef(false);

  useEffect(() => {
    if (initialized.current) return;
    initialized.current = true;
    client.startPolling();
    return () => {
      client.stopPolling().catch(() => {});
    };
  }, [client]);

  useEffect(() => {
    let disposed = false;
    let cleanup: (() => Promise<void> | void) | null = null;

    const pending = Promise.all([
      listen("baresync://data-changed", async () => {
        await queryClient.invalidateQueries({ queryKey: ["inventory"] });
        await queryClient.invalidateQueries({ queryKey: ["sync-state"] });
      }),
      listen("baresync://sync-status-changed", async () => {
        await queryClient.invalidateQueries({ queryKey: ["sync-state"] });
      }),
    ]).then(([unlistenDataChanged, unlistenStatusChanged]) => {
      const release = async () => {
        await Promise.all([unlistenDataChanged(), unlistenStatusChanged()]);
      };
      if (disposed) {
        release();
        return;
      }
      cleanup = release;
    });
    pending.catch(() => undefined);

    return () => {
      disposed = true;
      if (cleanup) cleanup();
    };
  }, [queryClient]);

  return (
    <SyncClientContext.Provider value={client}>
      {children}
    </SyncClientContext.Provider>
  );
}

export function useSyncClient(): SyncClient {
  const client = useContext(SyncClientContext);
  if (!client) {
    throw new Error("useSyncClient must be used within SyncClientProvider");
  }
  return client;
}
```

### 4. Query hook for Drizzle queries

```tsx
export function useDrizzleQuery<Row>(
  queryKey: QueryKey,
  buildQuery: () => PromiseLike<Row[]>
) {
  const query = useQuery({
    queryKey,
    queryFn: async () => await buildQuery(),
  });

  return {
    error: query.error ? String(query.error) : null,
    loading: query.isPending,
    refresh: async () => { await query.refetch(); },
    rows: query.data ?? [],
  };
}
```

### 5. Sync state hook

```tsx
import { useQuery } from "@tanstack/react-query";
import { useSyncClient } from "./useBaresyncQuery";

export function useSyncState() {
  const client = useSyncClient();
  const query = useQuery({
    queryKey: ["sync-state"],
    queryFn: async () => {
      const [localState, pollingStatus] = await Promise.all([
        client.getState(),
        client.getPollingStatus().catch(() => null),
      ]);
      return { localState, pollingStatus };
    },
  });

  return {
    localState: query.data?.localState ?? null,
    pollingStatus: query.data?.pollingStatus ?? null,
    loading: query.isPending,
  };
}
```

## Using in components

### Wire up the provider tree

```tsx
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { SyncClientProvider } from "./hooks/useBaresyncQuery";

const queryClient = new QueryClient();

function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <SyncClientProvider>
        <InventoryApp />
      </SyncClientProvider>
    </QueryClientProvider>
  );
}
```

### Query data

```tsx
import { useDrizzleQuery, useSyncClient } from "./hooks/useBaresyncQuery";
import { db, TABLE } from "./lib/db";

function ItemsList() {
  const items = useDrizzleQuery(["inventory", "items"], () =>
    db.select().from(TABLE.items)
  );

  if (items.loading) return <p>Loading...</p>;
  if (items.error) return <p>Error: {items.error}</p>;

  return (
    <ul>
      {items.rows.map((item) => (
        <li key={item.id}>{item.name}</li>
      ))}
    </ul>
  );
}
```

### Write data (insert)

```tsx
import { useSyncClient } from "./hooks/useBaresyncQuery";
import { db, TABLE } from "./lib/db";

function AddItemButton() {
  const client = useSyncClient();

  async function handleAdd() {
    const id = crypto.randomUUID();
    const timestamp = new Date().toISOString();

    await client.writeTransaction(db, async (tx) => {
      await client.writeLocalChange(tx, {
        operation: "insert",
        rowId: id,
        table: TABLE.items,
        write: (writeTx) =>
          writeTx.insert(TABLE.items).values({
            id,
            name: "New item",
            createdAt: timestamp,
            updatedAt: timestamp,
          }),
      });
    });
  }

  return <button onClick={handleAdd}>Add item</button>;
}
```

### Write data (soft delete)

```tsx
import { eq } from "drizzle-orm";

async function handleDelete(client: SyncClient, rowId: string) {
  const timestamp = new Date().toISOString();

  await client.writeTransaction(db, async (tx) => {
    await client.writeLocalChange(tx, {
      operation: "update",
      rowId,
      table: TABLE.items,
      write: (writeTx) =>
        writeTx
          .update(TABLE.items)
          .set({
            deletedAt: timestamp,
            isSynced: false,
            updatedAt: timestamp,
          })
          .where(eq(TABLE.items.id, rowId)),
    });
  });
}
```

### Sync status panel

```tsx
import { useState } from "react";
import { useSyncClient } from "./hooks/useBaresyncQuery";
import { useSyncState } from "./hooks/useSyncState";

function SyncPanel() {
  const client = useSyncClient();
  const { localState, pollingStatus } = useSyncState();
  const [syncing, setSyncing] = useState(false);

  async function handleSync() {
    setSyncing(true);
    try {
      await client.syncNow();
    } finally {
      setSyncing(false);
    }
  }

  return (
    <div>
      <button onClick={handleSync} disabled={syncing}>
        {syncing ? "Syncing..." : "Sync now"}
      </button>
      <p>Pending: {localState?.local_dirty_count ?? 0} rows</p>
      <p>
        Polling:{" "}
        {pollingStatus?.running
          ? pollingStatus.paused ? "Paused" : "Active"
          : "Stopped"}
      </p>
    </div>
  );
}
```

## Event flow summary

| Event | When emitted | What to invalidate |
|---|---|---|
| `baresync://data-changed` | Local data changed (insert, update, delete) | Data queries + sync state |
| `baresync://sync-status-changed` | Sync or polling status changed | Sync state only |

The `data-changed` event fires only when `rows_affected > 0` — it does not fire for no-op writes.

## Custom query keys

Replace `["inventory"]` and `["sync-state"]` with query keys that match your app. The key point is that `baresync://data-changed` should invalidate **all** your data query keys, and `baresync://sync-status-changed` should invalidate only your sync-state key.
