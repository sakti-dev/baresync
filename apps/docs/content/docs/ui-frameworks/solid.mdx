# Solid

This page shows the SolidJS equivalent of the React integration pattern. It uses `@tanstack/solid-query` for data fetching and Tauri events for cache invalidation.

Baresync does not ship a Solid-specific package. The integration follows the same event-driven pattern as React, using Solid's reactivity primitives instead of React hooks.

## Setup

### 1. Install dependencies

```bash
bun add @tanstack/solid-query baresync @tauri-apps/api drizzle-orm
```

### 2. Create the database

Same as React — this is framework-agnostic:

```tsx
import { invoke } from "@tauri-apps/api/core";
import { createTauriDrizzleDatabase } from "baresync/db";
import { items, locations } from "@your-app/sync-contract/local-synced-schema";
import { syncOutbox, syncCursors } from "@your-app/sync-contract/local-schema";

export const TABLE = { items, locations, syncOutbox, syncCursors };

export const db = createTauriDrizzleDatabase({ schema: TABLE, invoke });
```

### 3. Create the provider and context

Solid uses `createContext` and `createSignal` instead of React's `useState` and `useContext`. The event listener setup uses `onCleanup` for disposal.

```tsx
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";
import { createSyncClient, type SyncClient } from "baresync/tauri";
import {
  createContext,
  createEffect,
  createSignal,
  onCleanup,
  useContext,
  type ParentComponent,
} from "solid-js";
import { useQueryClient } from "@tanstack/solid-query";

const SyncClientContext = createContext<SyncClient>();

export const SyncClientProvider: ParentComponent = (props) => {
  const queryClient = useQueryClient();

  // Create the client once — createSignal with an initializer never re-creates it
  const [client] = createSignal(
    createSyncClient({
      scopeId: "your-scope-id",
      invoke,
    })
  );

  // Start background polling on mount, stop on cleanup
  createEffect(() => {
    client().startPolling();
    onCleanup(() => client().stopPolling().catch(() => {}));
  });

  // Subscribe to Tauri events emitted by the Rust sync engine.
  // These fire when observable local data changes or sync status changes.
  createEffect(() => {
    let disposed = false;
    let cleanup: (() => Promise<void> | void) | null = null;

    const pending = Promise.all([
      // Emitted after local writes, sync pull, or garbage collection
      // that actually changed rows (rows_affected > 0).
      // Invalidate both data queries and sync state so the UI updates.
      listen("baresync://data-changed", async () => {
        await queryClient.invalidateQueries({ queryKey: ["inventory"] });
        await queryClient.invalidateQueries({ queryKey: ["sync-state"] });
      }),
      // Emitted when polling starts/stops or a sync cycle completes.
      // Only need to refresh the sync indicator.
      listen("baresync://sync-status-changed", async () => {
        await queryClient.invalidateQueries({ queryKey: ["sync-state"] });
      }),
    ]).then(([unlistenDataChanged, unlistenStatusChanged]) => {
      const release = async () => {
        await Promise.all([unlistenDataChanged(), unlistenStatusChanged()]);
      };
      // If the owner was cleaned up before the listeners resolved, release immediately
      if (disposed) {
        release();
        return;
      }
      cleanup = release;
    });
    pending.catch(() => undefined);

    onCleanup(() => {
      disposed = true;
      if (cleanup) cleanup();
    });
  });

  return (
    <SyncClientContext.Provider value={client()}>
      {props.children}
    </SyncClientContext.Provider>
  );
};

export function useSyncClient(): SyncClient {
  const client = useContext(SyncClientContext);
  if (!client) {
    throw new Error("useSyncClient must be used within SyncClientProvider");
  }
  return client;
}
```

### 4. Query helper for Drizzle queries

Solid Query's `createQuery` returns a signal-based result. `Row` is inferred from your Drizzle table schema using `$inferSelect`:

```ts
type ItemRow = typeof TABLE.items.$inferSelect;
// { id: string; name: string; scopeId: string; updatedAt: string; ... }
```

Wrap the query for a consistent accessor-based API:

```tsx
import { createQuery } from "@tanstack/solid-query";

interface UseQueryAccessors<Row> {
  error: () => string | null;
  loading: () => boolean;
  rows: () => Row[];
}

export function useDrizzleQuery<Row>(
  queryKey: unknown[],
  buildQuery: () => Promise<Row[]>
): UseQueryAccessors<Row> {
  const query = createQuery(() => ({
    queryKey,
    queryFn: async () => await buildQuery(),
  }));

  return {
    error: () => (query.error ? String(query.error) : null),
    loading: () => query.isPending,
    rows: () => query.data ?? [],
  };
}
```

TypeScript infers `Row` from the `buildQuery` callback — you don't pass it explicitly. The returned properties are **accessor functions** (signals), not plain values.

```ts
// Row is inferred as ItemRow automatically
const items = useDrizzleQuery(["inventory", "items"], () =>
  db.select().from(TABLE.items)
);
```

### 5. Sync state helper

```tsx
import { createQuery } from "@tanstack/solid-query";
import { useSyncClient } from "./useBaresyncQuery";

export function useSyncState() {
  const client = useSyncClient();

  const query = createQuery(() => ({
    queryKey: ["sync-state"],
    queryFn: async () => {
      const [localState, pollingStatus] = await Promise.all([
        client.getState(),
        client.getPollingStatus().catch(() => null),
      ]);
      return { localState, pollingStatus };
    },
  }));

  return {
    localState: () => query.data?.localState ?? null,
    pollingStatus: () => query.data?.pollingStatus ?? null,
    loading: () => query.isPending,
  };
}
```

## Using in components

### Wire up the provider tree

```tsx
import { QueryClient, QueryClientProvider } from "@tanstack/solid-query";
import { SyncClientProvider } from "./hooks/useBaresyncQuery";

const queryClient = new QueryClient();

function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <SyncClientProvider>
        <InventoryApp />
      </SyncClientProvider>
    </QueryClientProvider>
  );
}
```

### Query data

```tsx
import { For, Show } from "solid-js";
import { useDrizzleQuery } from "./hooks/useBaresyncQuery";
import { db, TABLE } from "./lib/db";

function ItemsList() {
  const items = useDrizzleQuery(["inventory", "items"], () =>
    db.select().from(TABLE.items)
  );

  return (
    <>
      <Show when={items.loading()}>
        <p>Loading...</p>
      </Show>
      <Show when={items.error()}>
        <p>Error: {items.error()}</p>
      </Show>
      <ul>
        <For each={items.rows()}>
          {(item) => <li>{item.name}</li>}
        </For>
      </ul>
    </>
  );
}
```

### Write data (insert)

```tsx
import { useQueryClient } from "@tanstack/solid-query";
import { useSyncClient } from "./hooks/useBaresyncQuery";
import { db, TABLE } from "./lib/db";

function AddItemButton() {
  const client = useSyncClient();
  const queryClient = useQueryClient();

  async function handleAdd() {
    const id = crypto.randomUUID();
    const timestamp = new Date().toISOString();

    // writeTransaction wraps all changes in a single SQLite transaction.
    // writeLocalChange performs the Drizzle write AND creates an outbox entry
    // so the sync engine picks it up on the next push.
    await client.writeTransaction(db, async (tx) => {
      await client.writeLocalChange(tx, {
        operation: "insert",
        rowId: id,
        table: TABLE.items,
        write: (writeTx) =>
          writeTx.insert(TABLE.items).values({
            id,
            name: "New item",
            createdAt: timestamp,
            updatedAt: timestamp,
          }),
      });
    });

    // Eagerly invalidate so the user sees the new row immediately.
    // The baresync://data-changed event will also fire, but it may arrive
    // slightly later — this ensures instant feedback.
    await queryClient.invalidateQueries({ queryKey: ["inventory", "items"] });
  }

  return <button onClick={handleAdd}>Add item</button>;
}
```

### Write data (soft delete)

```tsx
import { useQueryClient } from "@tanstack/solid-query";
import { eq } from "drizzle-orm";
import { db, TABLE } from "./lib/db";

async function handleDelete(
  client: SyncClient,
  queryClient: QueryClient,
  rowId: string,
) {
  const timestamp = new Date().toISOString();

  // Soft delete: set deletedAt and mark unsynced so the engine pushes the deletion.
  // The row stays in the table until garbage collection hard-deletes it after sync.
  await client.writeTransaction(db, async (tx) => {
    await client.writeLocalChange(tx, {
      operation: "update",
      rowId,
      table: TABLE.items,
      write: (writeTx) =>
        writeTx
          .update(TABLE.items)
          .set({
            deletedAt: timestamp,
            isSynced: false,
            updatedAt: timestamp,
          })
          .where(eq(TABLE.items.id, rowId)),
    });
  });

  // Refresh the list so the deleted row disappears (or shows as struck-through)
  await queryClient.invalidateQueries({ queryKey: ["inventory", "items"] });
}
```

### Sync status panel

```tsx
import { createSignal } from "solid-js";
import { useSyncClient } from "./hooks/useBaresyncQuery";
import { useSyncState } from "./hooks/useSyncState";

function SyncPanel() {
  const client = useSyncClient();
  const { localState, pollingStatus } = useSyncState();
  const [syncing, setSyncing] = createSignal(false);

  async function handleSync() {
    setSyncing(true);
    try {
      await client.syncNow();
    } finally {
      setSyncing(false);
    }
  }

  return (
    <div>
      <button onClick={handleSync} disabled={syncing()}>
        {syncing() ? "Syncing..." : "Sync now"}
      </button>
      <p>Pending: {localState()?.local_dirty_count ?? 0} rows</p>
      <p>
        Polling:{" "}
        {pollingStatus()?.running
          ? pollingStatus()?.paused ? "Paused" : "Active"
          : "Stopped"}
      </p>
    </div>
  );
}
```

## Key differences from React

| Concern | React | Solid |
|---|---|---|
| State primitive | `useState` | `createSignal` |
| Side effects | `useEffect` | `createEffect` + `onCleanup` |
| Context | `createContext` + `useContext` | `createContext` + `useContext` |
| Query result | Plain object `{ rows, loading }` | Accessor functions `rows()`, `loading()` |
| List rendering | `map()` | `<For each={}>` |
| Conditional | `&&` or ternary | `<Show when={}>` |
| Provider type | Function component | `ParentComponent` |
| Query hook | `useQuery` | `createQuery` (returns a reactive object) |

The write patterns (`writeTransaction`, `writeLocalChange`) are identical — they don't use any framework primitives.
