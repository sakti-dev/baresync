# JS Client Tests

Baresync's JS APIs are designed for testability — you inject a mock `invoke` function and everything works without Tauri or a real server.

## Mocking the sync client

Pass a mock `invoke` to `createSyncClient`. It records which commands your code calls and returns whatever data you want:

```ts
import { createSyncClient } from "baresync/tauri";
import { describe, expect, it, vi } from "vitest";

function createMockClient(responses: Record<string, unknown> = {}) {
  return createSyncClient({
    apiUrl: "https://api.example.com",
    encoding: "json",
    scopeId: "merchant-1",
    invoke: (cmd) => Promise.resolve(responses[cmd] ?? {}),
  });
}

it("calls getState and reads dirty count", async () => {
  const client = createMockClient({
    get_sync_local_state: {
      local_dirty_count: 3,
      last_server_watermark: "",
      needs_baseline_sync: false,
    },
  });

  const state = await client.getState();
  expect(state.local_dirty_count).toBe(3);
});
```

### Verifying commands are called

Record every call to assert your code triggers the right sync operations:

```ts
it("triggers syncNow after form submit", async () => {
  const calls: string[] = [];
  const client = createSyncClient({
    apiUrl: "https://api.example.com",
    encoding: "json",
    scopeId: "merchant-1",
    invoke: (cmd) => { calls.push(cmd); return Promise.resolve({}); },
  });

  await submitForm(client); // your code under test

  expect(calls).toContain("sync_now");
});
```

### Simulating errors

Reject `invoke` to test error handling in your UI:

```ts
it("shows error banner when sync fails", async () => {
  const client = createSyncClient({
    apiUrl: "https://api.example.com",
    encoding: "json",
    scopeId: "merchant-1",
    invoke: () => Promise.reject(new Error("network error")),
  });

  await expect(client.syncNow()).rejects.toThrow("network error");
});
```

## Testing write patterns

Mock the database transaction to verify your writes create outbox entries:

```ts
function createRecordingTx() {
  const inserted: Array<{ table: unknown; values: Record<string, unknown> }> = [];
  const tx = {
    insert(table: unknown) {
      return {
        values(values: Record<string, unknown>) {
          inserted.push({ table, values });
          return Promise.resolve({ rowsAffected: 1 });
        },
      };
    },
  };
  return { inserted, tx };
}

it("writeLocalChange creates an outbox entry for the right table", async () => {
  const { inserted, tx } = createRecordingTx();
  const client = createSyncClient({
    apiUrl: "https://api.example.com",
    encoding: "json",
    scopeId: "merchant-1",
    invoke: () => Promise.resolve({}),
  });

  await client.writeLocalChange(tx, {
    operation: "insert",
    rowId: "item-1",
    table: myItemsTable,
    write(writeTx) {
      // your insert logic — the mock tx records it
    },
  });

  // One outbox entry was created with the right table and row
  expect(inserted).toHaveLength(1);
  expect(inserted[0].values.tableName).toBe("items");
  expect(inserted[0].values.rowId).toBe("item-1");
});
```

## Testing React hooks and providers

Mock `baresync/tauri` and `@tauri-apps/api/event` to test your sync provider and hooks:

```ts
import { render, waitFor } from "@testing-library/react";
import { createElement } from "react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { afterEach, describe, expect, it, vi } from "vitest";

const listen = vi.hoisted(() => vi.fn());
const createSyncClient = vi.hoisted(() => vi.fn());

vi.mock("baresync/tauri", () => ({ createSyncClient }));
vi.mock("@tauri-apps/api/event", () => ({ listen }));

it("invalidates queries when data changes", async () => {
  const queryClient = new QueryClient();
  const invalidate = vi.spyOn(queryClient, "invalidateQueries");

  createSyncClient.mockReturnValue({
    startPolling: vi.fn(),
    stopPolling: vi.fn().mockResolvedValue(undefined),
  });

  // Capture the event listener your provider registers
  const listeners = new Map<string, (event: unknown) => void>();
  listen.mockImplementation((event, handler) => {
    listeners.set(event, handler);
    return Promise.resolve(async () => {});
  });

  render(
    createElement(
      QueryClientProvider,
      { client: queryClient },
      createElement(SyncClientProvider, null, createElement("div", null, "child"))
    )
  );

  // Wait for the provider to register its listener
  await waitFor(() => {
    expect(listeners.has("baresync://data-changed")).toBe(true);
  });

  // Simulate the data-changed event from the plugin
  await listeners.get("baresync://data-changed")?.(undefined);

  expect(invalidate).toHaveBeenCalledWith({ queryKey: ["inventory"] });
});
```

## Testing components that display sync state

```ts
import { render, waitFor } from "@testing-library/react";

it("shows sync status from getState", async () => {
  const client = createSyncClient({
    apiUrl: "https://api.example.com",
    encoding: "json",
    scopeId: "merchant-1",
    invoke: (cmd) => {
      if (cmd === "get_sync_local_state") {
        return Promise.resolve({
          local_dirty_count: 5,
          needs_baseline_sync: false,
        });
      }
      return Promise.resolve({});
    },
  });

  // Your component that calls client.getState() and renders the result
  const { findByText } = render(createElement(SyncIndicator, { client }));

  expect(await findByText("5 pending")).toBeDefined();
});
```

## Mocking the Drizzle proxy

When testing code that queries through `createTauriDrizzleDatabase`, mock `invoke` to return empty results:

```ts
import { createTauriDrizzleDatabase } from "baresync/db";
import { describe, expect, test } from "bun:test";

const mockDb = createTauriDrizzleDatabase({
  schema: TABLE,
  invoke: async (cmd, _args) => {
    if (cmd === "run_sql") return []; // empty rows for SELECT
    return { last_insert_id: 0, rows_affected: 0 }; // for INSERT/UPDATE
  },
});
```

For tests that verify SQL generation, record the calls:

```ts
const calls: Array<{ sql: string }> = [];
const db = createTauriDrizzleDatabase({
  schema: TABLE,
  invoke: async (cmd, args) => {
    if (cmd === "run_sql") {
      calls.push(args as { sql: string });
      return [];
    }
    return { last_insert_id: 0, rows_affected: 0 };
  },
});

await db.select().from(TABLE.items);
expect(calls[0].sql).toContain("items");
```

## Framework choice

| Test framework | When to use |
|---|---|
| **Vitest** (default) | Most tests — hooks, components, client mocking |
| **bun:test** | Tests that need `bun:sqlite` for real SQL (server repository tests) |

Both are auto-discovered by `bun test`.
