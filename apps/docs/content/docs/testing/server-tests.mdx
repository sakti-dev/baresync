# Server Tests

Baresync's server handlers are testable functions — you create a handler with mock callbacks and send it real `Request` objects. No running server needed.

## Testing push handlers

Create a handler with a mock `resolveScope` and `applyPushChanges`, then send it a request:

```ts
import { createSyncPushHandler } from "baresync/server";
import { describe, expect, it, vi } from "vitest";

it("applies authorized push changes in table order", async () => {
  const appliedChanges: unknown[] = [];

  const handler = createSyncPushHandler({
    encoding: "json",
    resolveScope: vi.fn(async ({ scopeId }) => ({
      ok: true,
      scope: { merchantId: scopeId },
    })),
    applyPushChanges: vi.fn(async (input) => ({
      acceptedTables: input.changes.map((c) => c.table),
      scopeId: input.scopeId,
      serverTime: "2026-05-20T00:00:00.000Z",
    })),
    idempotency: { db: createTestDb() },
    upsertOrder: ["categories", "products"],
  });

  const response = await handler(
    new Request("http://localhost/sync", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        scopeId: "merchant-1",
        clientId: "client-1",
        idempotencyKey: "idem-1",
        tables: [
          { table: "products", changedRows: [{ id: "p1" }], deletedIds: [] },
          { table: "categories", changedRows: [{ id: "c1" }], deletedIds: [] },
        ],
      }),
    }),
    { sessionId: "session-1" } // your request context
  );

  expect(response.status).toBe(200);
  const body = await response.json();
  expect(body.acceptedTables).toEqual(["categories", "products"]); // reordered per upsertOrder
});
```

### Testing authorization rejection

When `resolveScope` returns `{ ok: false }`, the handler short-circuits without calling `applyPushChanges`:

```ts
it("returns 403 for unauthorized scope", async () => {
  const applyPushChanges = vi.fn();

  const handler = createSyncPushHandler({
    encoding: "json",
    resolveScope: vi.fn(async () => ({
      ok: false,
      status: 403,
      body: { error: "forbidden" },
    })),
    applyPushChanges,
    idempotency: { db: createTestDb() },
    upsertOrder: ["categories", "products"],
  });

  const response = await handler(createJsonRequest({ ... }), { sessionId: "s1" });

  expect(response.status).toBe(403);
  expect(applyPushChanges).not.toHaveBeenCalled();
});
```

## Testing pull and status handlers

Same pattern — mock `loadPullChanges` or `loadSyncStatus`:

```ts
it("returns pull data for authorized scope", async () => {
  const handler = createSyncPullHandler({
    encoding: "json",
    resolveScope: vi.fn(async ({ scopeId }) => ({
      ok: true,
      scope: { merchantId: scopeId },
    })),
    loadPullChanges: vi.fn(async (input) => ({
      cursor: input.cursor,
      hasMore: false,
      serverTime: "2026-05-20T00:00:00.000Z",
      tables: input.tables.map((table) => ({
        table,
        changedRows: [],
        deletedIds: [],
        acceptedCreatedIds: [],
        acceptedUpdatedIds: [],
        acceptedDeletedIds: [],
        rejected: [],
      })),
    })),
    limit: 25,
  });

  const response = await handler(
    createJsonRequest({
      scopeId: "merchant-1",
      cursor: "sync:123:categories:row-1",
      tables: ["categories"],
    }),
    { sessionId: "session-1" }
  );

  expect(response.status).toBe(200);
});
```

## Testing idempotency

Use an in-memory SQLite database to test that duplicate pushes are handled correctly:

```ts
import { createIdempotencyGuard } from "baresync/server";
import { Database } from "bun:sqlite";
import { drizzle } from "drizzle-orm/bun-sqlite";

function createTestDb() {
  const sqlite = new Database(":memory:");
  sqlite.run("PRAGMA journal_mode = WAL");
  sqlite.exec(`CREATE TABLE sync_batch_requests (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    client_id TEXT NOT NULL,
    idempotency_key TEXT NOT NULL,
    request_hash TEXT NOT NULL,
    status TEXT NOT NULL,
    response_body TEXT,
    created_at INTEGER NOT NULL,
    completed_at INTEGER,
    UNIQUE(client_id, idempotency_key)
  )`);
  return drizzle(sqlite) as unknown as SqliteRemoteDatabase;
}

it("replays cached response on duplicate push", async () => {
  const db = createTestDb();
  const guard = createIdempotencyGuard({ db });
  let callCount = 0;

  const first = await guard.run(
    { clientId: "client-1", idempotencyKey: "key-1", requestHash: "hash-a" },
    async () => {
      callCount++;
      return { serverTime: "2026-05-19T12:00:00.000Z" };
    }
  );

  const second = await guard.run(
    { clientId: "client-1", idempotencyKey: "key-1", requestHash: "hash-a" },
    async () => {
      callCount++;
      return { serverTime: "2026-05-19T12:00:00.000Z" };
    }
  );

  expect(first.wasReplay).toBe(false);
  expect(second.wasReplay).toBe(true);
  expect(callCount).toBe(1); // callback only ran once
  expect(first.result).toEqual(second.result);
});
```

## Testing the full server pipeline

Chain `decodeSyncRequest` → `validatePushEnvelope` → `orderPushChanges` → idempotency guard → `encodeSyncResponse` to test your pipeline end-to-end:

```ts
import {
  decodeSyncRequest,
  validatePushEnvelope,
  orderPushChanges,
  encodeSyncResponse,
} from "baresync/server";

it("processes a push through the full pipeline", async () => {
  const db = createTestDb();
  const guard = createIdempotencyGuard({ db });

  const pushBody = {
    scopeId: "merchant-1",
    clientId: "client-1",
    idempotencyKey: "idem-001",
    tables: [
      { table: "categories", changedRows: [{ id: "cat-1" }], deletedIds: [] },
    ],
  };

  const request = new Request("http://localhost/sync", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(pushBody),
  });

  const decoded = await decodeSyncRequest({ encoding: "json", kind: "push", request });

  validatePushEnvelope(decoded, { maxBytes: 1024 * 1024, maxRows: 1000 });

  const ordered = orderPushChanges({
    changes: pushBody.tables,
    order: ["categories", "products"],
  });

  const { result } = await guard.run(
    {
      clientId: pushBody.clientId,
      idempotencyKey: pushBody.idempotencyKey,
      requestHash: decoded.requestHash,
    },
    async () => ({ serverTime: "2026-05-19T12:00:00.000Z" })
  );

  const response = encodeSyncResponse({ body: result, encoding: "json", kind: "push" });
  expect(response.status).toBe(200);
});
```

## Testing cleanup

```ts
import { cleanupSyncBatchRequests } from "baresync/server";

it("deletes old completed requests", async () => {
  const db = createTestDb();
  // ... insert old and recent batch requests ...

  const result = await cleanupSyncBatchRequests({
    db,
    olderThanMs: 7 * 24 * 60 * 60 * 1000, // 7 days
  });

  expect(result.deletedCount).toBe(1);
});
```
